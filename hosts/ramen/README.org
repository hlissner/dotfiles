* Filesystems
** =/dev/nvme0n1= (PC401 NVMe SK Hynix 512gb) :linux:
#+begin_src sh
parted /dev/nvme0n1 -- mklabel gpt
parted /dev/nvme0n1 -- mkpart ESP fat32 1MiB 1GiB
parted /dev/nvme0n1 -- mkpart primary 1GiB 256GiB
parted /dev/nvme0n1 -- mkpart primary 256GiB 100%
parted /dev/nvme0n1 -- set 1 boot on
mkfs.fat -F32 -n BOOT /dev/nvme0n1p1
mkfs.ext4 -L nixos /dev/nvme0n1p2

# Encrypt /home in case my laptop is ever stolen.
cryptsetup luksFormat --label home-crypt /dev/nvme0n1p3
cryptsetup open /dev/disk/by-label/home-crypt home
mkfs.ext4 -L home /dev/mapper/home

mount /dev/disk/by-label/nixos /mnt
mkdir -p /mnt/{home,boot}
mount /dev/disk/by-label/BOOT /mnt/boot
mount /dev/disk/by-label/home /mnt/home
#+end_src
