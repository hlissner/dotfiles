* Filesystems
** =/dev/nvme0n1= (Samsung EVO+ 500gb) :linux:
#+begin_src sh
parted /dev/nvme0n1 -- mklabel gpt
parted /dev/nvme0n1 -- mkpart ESP fat32 1MiB 512MiB
parted /dev/nvme0n1 -- mkpart primary 512MiB 200GiB
parted /dev/nvme0n1 -- mkpart primary 200GiB 100%
parted /dev/nvme0n1 -- set 1 boot on
mkfs.fat -F32 -n BOOT /dev/nvme0n1p1
mkfs.ext4 -L nixos /dev/nvme0n1p2
mkfs.ext4 -L home /dev/nvme0n1p3

mount /dev/disk/by-label/nixos /mnt
mkdir -p /mnt/{home,boot}
mount /dev/disk/by-label/BOOT /mnt/boot
mount /dev/disk/by-label/home /mnt/home
#+end_src

** =/dev/nvme1n1= (SN850X NVMe 1TB)
#+begin_src sh
parted /dev/nvme1n1 -- mklabel gpt
parted /dev/nvme1n1 -- mkpart primary 0% 100%
mkfs.ext4 -L data /dev/nvme1n1

mkdir -p /mnt/media/data
mount /dev/disk/by-label/data /mnt/media/data
#+end_src

** =/dev/sda= (WD6400AAKS 640gb)
#+begin_src sh
parted /dev/sda -- mklabel gpt
parted /dev/sda -- mkpart primary 0% 100%
mkfs.ext4 -L data1 /dev/sda1

mkdir -p /mnt/media/data1
mount /dev/disk/by-label/data1 /mnt/media/data1
#+end_src

** =/dev/sdb= (ST1000DM010 1TB)
#+begin_src sh
parted /dev/sdb -- mklabel gpt
parted /dev/sdb -- mkpart primary 0% 100%
mkfs.ext4 -L backup /dev/sdb1

mkdir -p /mnt/media/backup
mount /dev/disk/by-label/backup /mnt/media/backup
#+end_src
